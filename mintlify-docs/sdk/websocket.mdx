## Overview

The SDK supports real-time quote streaming via WebSocket connections for:
- Live price monitoring
- Real-time quote updates
- Streaming market data

## Basic Usage

### Creating a Quote Stream

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

const client = new KaleidoClient({
  baseUrl: 'https://api.regtest.kaleidoswap.com',
});

// Create stream
const stream = await client.createQuoteStream('BTC/USDT');

try {
  for (let i = 0; i < 5; i++) {
    // Receive with timeout (returns null on timeout)
    const quoteJson = await stream.recv(10.0);

    if (quoteJson === null) {
      console.log('Timeout - no quote received');
      break;
    }

    const quote = JSON.parse(quoteJson);
    console.log(`Quote ${i + 1}: Price ${quote.price}`);
  }
} finally {
  // Always close the stream
  stream.close();
}
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig

config = KaleidoConfig(base_url="https://api.regtest.kaleidoswap.com")
client = KaleidoClient(config)

# Using context manager (recommended)
with client.subscribe_quotes("BTC/USDT") as stream:
    for quote in stream.iter_quotes(timeout=10.0, max_quotes=5):
        print(f"Price: {quote.price}")
```
</CodeGroup>

## Stream API

### TypeScript: IQuoteStream

```typescript
interface IQuoteStream {
  // Receive the next quote (blocking with timeout)
  // Returns null on timeout
  recv(timeoutSecs: number): Promise<string | null>;

  // Check if stream is connected
  isConnected(): boolean;

  // Close the stream
  close(): void;
}
```

### Python: QuoteStreamContext

```python
class QuoteStreamContext:
    # Receive and parse the next quote
    def recv(self, timeout: float = 5.0) -> Optional[PairQuoteResponse]
    
    # Iterate over quotes
    def iter_quotes(
        self, 
        timeout: float = 5.0, 
        max_quotes: Optional[int] = None
    ) -> Iterator[PairQuoteResponse]
    
    # Check if connected
    def is_connected(self) -> bool
```

## Price Monitoring

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

async function monitorPrices() {
  const client = new KaleidoClient({
    baseUrl: 'https://api.regtest.kaleidoswap.com',
  });

  const prices: Record<string, number> = {};

  const stream = await client.createQuoteStream('BTC/USDT');

  try {
    for (let i = 0; i < 10; i++) {
      const quoteJson = await stream.recv(10.0);
      if (!quoteJson) break;

      const quote = JSON.parse(quoteJson);
      const oldPrice = prices['BTC/USDT'];
      const newPrice = parseFloat(quote.price);
      prices['BTC/USDT'] = newPrice;

      if (oldPrice) {
        const change = ((newPrice - oldPrice) / oldPrice) * 100;
        const direction = change > 0 ? '📈' : change < 0 ? '📉' : '➡️';
        console.log(`${direction} ${newPrice.toFixed(2)} (${change.toFixed(2)}%)`);
      } else {
        console.log(`💰 ${newPrice.toFixed(2)}`);
      }
    }
  } finally {
    stream.close();
  }
}

monitorPrices();
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig

                print(f"{direction} {new_price:.2f} ({change:+.2f}%)")
            else:
                print(f"💰 {new_price:.2f}")

if __name__ == "__main__":
    monitor_prices()
```

## Trading Signal Generator

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

async function generateSignals() {
  const client = new KaleidoClient({
    baseUrl: 'https://api.regtest.kaleidoswap.com',
  });

  const priceHistory: number[] = [];
  const maxHistorySize = 10;

  console.log('Generating signals for BTC/USDT...\n');

  const stream = await client.createQuoteStream('BTC/USDT');

  try {
    for (let i = 0; i < 20; i++) {
      const quoteJson = await stream.recv(10.0);
      if (!quoteJson) break;

      const quote = JSON.parse(quoteJson);
      const price = parseFloat(quote.price);

      priceHistory.push(price);
      if (priceHistory.length > maxHistorySize) {
        priceHistory.shift();
      }

      if (priceHistory.length < 5) {
        console.log(`Price: ${price.toFixed(2)} (collecting...)`);
        continue;
      }

      // Simple moving average
      const sma = priceHistory.reduce((a, b) => a + b, 0) / priceHistory.length;

      // Generate signal
      let signal: string;
      if (price > sma * 1.02) signal = '🔴 SELL';
      else if (price < sma * 0.98) signal = '🟢 BUY';
      else signal = '⚪ HOLD';

      console.log(`Price: ${price.toFixed(2)} | SMA: ${sma.toFixed(2)} | ${signal}`);
    }
  } finally {
    stream.close();
  }
}

```

```python Python
from collections import deque
from kaleidoswap import KaleidoClient, KaleidoConfig

def generate_signals():
    config = KaleidoConfig(base_url="https://api.regtest.kaleidoswap.com")
    client = KaleidoClient(config)
    
    price_history = deque(maxlen=10)
    
    print("Generating signals for BTC/USDT...\n")
    
    with client.subscribe_quotes("BTC/USDT") as stream:
        for quote in stream.iter_quotes(timeout=10.0, max_quotes=20):
            price = float(quote.price)
            price_history.append(price)
            
            if len(price_history) < 5:
                print(f"Price: {price:.2f} (collecting...)")
                continue
            
            # Simple moving average
            sma = sum(price_history) / len(price_history)
            
            # Generate signal
            if price > sma * 1.02:
                signal = "🔴 SELL"
            elif price < sma * 0.98:
                signal = "🟢 BUY"
            else:
                signal = "⚪ HOLD"
            
            print(f"Price: {price:.2f} | SMA: {sma:.2f} | {signal}")

if __name__ == "__main__":
    generate_signals()
```

## Quote Aggregation

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

interface QuoteStats {
  min: number;
  max: number;
  avg: number;
  count: number;
}

async function aggregateQuotes(): Promise<QuoteStats | null> {
  const client = new KaleidoClient({
    baseUrl: 'https://api.regtest.kaleidoswap.com',
  });

  const prices: number[] = [];

  const stream = await client.createQuoteStream('BTC/USDT');

  try {
    for (let i = 0; i < 10; i++) {
      const quoteJson = await stream.recv(10.0);
      if (!quoteJson) break;

      const quote = JSON.parse(quoteJson);
      prices.push(parseFloat(quote.price));
    }
  } finally {
    stream.close();
  }

  if (prices.length === 0) return null;

  return {
    min: Math.min(...prices),
    max: Math.max(...prices),
    avg: prices.reduce((a, b) => a + b, 0) / prices.length,
    count: prices.length,
  };
}

aggregateQuotes().then(stats => {
  if (stats) {
    console.log(`📊 Stats (${stats.count} quotes)`);
    console.log(`   Min: ${stats.min.toFixed(2)}`);
    console.log(`   Max: ${stats.max.toFixed(2)}`);
    console.log(`   Avg: ${stats.avg.toFixed(2)}`);
  }
});
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig

def aggregate_quotes():
    config = KaleidoConfig(base_url="https://api.regtest.kaleidoswap.com")
    client = KaleidoClient(config)
    
    prices = []
    
    with client.subscribe_quotes("BTC/USDT") as stream:
        for quote in stream.iter_quotes(timeout=10.0, max_quotes=10):
            prices.append(float(quote.price))
    
    if not prices:
        print("No quotes received")
        return
    
    print(f"📊 Stats ({len(prices)} quotes)")
    print(f"   Min: {min(prices):.2f}")
    print(f"   Max: {max(prices):.2f}")
    print(f"   Avg: {sum(prices) / len(prices):.2f}")

if __name__ == "__main__":
    aggregate_quotes()
```

## Error Handling

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, KaleidoError } from '@kaleidoswap/sdk';

async function robustStreaming() {
  const client = new KaleidoClient({
    baseUrl: 'https://api.regtest.kaleidoswap.com',
  });

  let stream;

  try {
    stream = await client.createQuoteStream('BTC/USDT');

    while (true) {
      if (!stream.isConnected()) {
        console.log('Stream disconnected');
        break;
      }

      const quoteJson = await stream.recv(10.0);

      if (quoteJson === null) {
        console.log('Timeout');
        continue;
      }

      const quote = JSON.parse(quoteJson);
      console.log(`Quote: ${quote.price}`);
    }
  } catch (error) {
    if (error instanceof KaleidoError) {
      console.log(`SDK error: ${error.message}`);
    } else {
      console.log(`Unexpected error: ${error}`);
    }
  } finally {
    if (stream) stream.close();
  }
}

robustStreaming();
```


## Best Practices

<AccordionGroup>
  <Accordion title="Always close streams">
    Use try/finally in TypeScript or context managers in Python to ensure streams are properly closed.
  </Accordion>
  
  <Accordion title="Check connection status">
    Use `isConnected()` / `is_connected()` for long-running streams to detect disconnections.
  </Accordion>
  
  <Accordion title="Set appropriate timeouts">
    Choose timeouts based on expected quote frequency to prevent hanging.
  </Accordion>
  
  <Accordion title="Handle null returns">
    `recv()` returns null on timeout - always check before parsing.
  </Accordion>
  
  <Accordion title="Implement reconnection logic">
    For production use, implement automatic reconnection on stream failures.
  </Accordion>
</AccordionGroup>
