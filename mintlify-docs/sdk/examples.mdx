```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

async function exploreMarket() {

}
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig

        if pair.routes:
            route = pair.routes[0]
            print(f"    Min: {route.min_from_amount}")
            print(f"    Max: {route.max_from_amount}")

if __name__ == "__main__":
```

## Complete Atomic Swap

This example demonstrates the full atomic swap workflow:

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, Layer } from '@kaleidoswap/sdk';

async function atomicSwap() {
  const client = await KaleidoClient.create({
    baseUrl: process.env.KALEIDO_API_URL || 'http://localhost:8000',
    nodeUrl: process.env.KALEIDO_NODE_URL || 'http://localhost:3001',
  });

  console.log('=== Atomic Swap Flow ===\n');

  if (!client.hasNode()) {
    console.log('⚠️ Node not configured');
    return;
  }

  // Step 1: Get trading pairs
  console.log('📋 Step 1: Getting pairs...');
  const pairs = await client.listPairs();
  if (pairs.length === 0) return;

  const pair = pairs[0];
  const pairTicker = `${pair.baseAsset}/${pair.quoteAsset}`;
  console.log(`   Using: ${pairTicker}`);

  // Step 2: Get a quote
  console.log('\n💱 Step 2: Getting quote...');
  const amount = pair.minBaseOrderSize ? BigInt(pair.minBaseOrderSize) : 10_000_000n;
  const quote = await client.getQuoteByPair(pairTicker, amount);
  console.log(`   Price: ${quote.price}`);

  // Step 3: Get taker node info
  console.log('\n🔑 Step 3: Getting node info...');
  const nodeInfo = await client.getNodeInfo();
  console.log(`   Pubkey: ${nodeInfo.pubkey.substring(0, 20)}...`);

  // Step 4: Create Swap Order
  console.log('\n🔄 Step 4: Creating swap order...');
  const order = await client.createSwapOrder({
    from_asset: quote.fromAsset.assetId,
    to_asset: quote.toAsset.assetId,
    from_amount: quote.fromAmount, // BigInt
    to_amount: quote.toAmount,     // BigInt
    taker_pubkey: nodeInfo.pubkey,
    // Define layers
    from_layer: Layer.BTC_LN,
    to_layer: Layer.RGB_LN,
  });

  console.log(`   Order ID: ${order.id}`);
  console.log(`   Payment Request: ${order.paymentRequest}`);

  // Step 5: Wait for completion
  console.log('\n⏳ Step 5: Waiting for completion...');
  // Assuming payment is made externally or via node wallet hook if integrated
  // Here we just monitor status
  const status = await client.getSwapOrderStatus(order.id);
  console.log(`   Final Status: ${status.status}`);
}

atomicSwap().catch(console.error);
```

```python Python
import json
import os
from kaleidoswap import KaleidoClient, KaleidoConfig

def atomic_swap():
    config = KaleidoConfig(
        base_url=os.environ.get("KALEIDO_API_URL", "http://localhost:8000"),
        node_url=os.environ.get("KALEIDO_NODE_URL", "http://localhost:3001"),
    )
    client = KaleidoClient(config)
    
    print("=== Atomic Swap Flow ===\n")
    
    if not client.has_node():
        print("⚠️ Node not configured")
        return
    
    # Step 1: Get trading pairs
    print("📋 Step 1: Getting pairs...")
    pairs = client.list_pairs()
    if not pairs:
        return
    
    pair = pairs[0]
    base = pair.base.ticker if pair.base else "BTC"
    quote = pair.quote.ticker if pair.quote else "USDT"
    pair_ticker = f"{base}/{quote}"
    print(f"   Using: {pair_ticker}")
    
    # Step 2: Get a quote
    print("\n💱 Step 2: Getting quote...")
    amount = 10_000_000
    if pair.routes and pair.routes[0].min_from_amount:
        amount = pair.routes[0].min_from_amount
    
    quote_resp = client.get_quote_by_pair(pair_ticker, from_amount=amount)
    print(f"   RFQ ID: {quote_resp.rfq_id}")
    print(f"   Price: {quote_resp.price}")
    
    # Step 3: Get taker node info
    print("\n🔑 Step 3: Getting node info...")
    node_info_json = client.get_rgb_node_info()
    node_info = json.loads(node_info_json)
    print(f"   Pubkey: {node_info['pubkey'][:20]}...")
    
    # Step 4: Initialize swap
    print("\n🔄 Step 4: Initializing swap...")
    init_result = client.init_swap({
        "rfq_id": quote_resp.rfq_id,
        "from_asset": quote_resp.from_asset.asset_id,
        "from_amount": quote_resp.from_asset.amount,
        "to_asset": quote_resp.to_asset.asset_id,
        "to_amount": quote_resp.to_asset.amount,
    })
    init_data = json.loads(init_result)
    swapstring = init_data["swapstring"]
    payment_hash = init_data["payment_hash"]
    print(f"   Payment hash: {payment_hash}")
    
    # Step 5: Whitelist trade
    print("\n🔐 Step 5: Whitelisting...")
    client.whitelist_trade(swapstring)
    print("   ✅ Done")
    
    # Step 6: Execute swap
    print("\n⚡ Step 6: Executing...")
    client.execute_swap({
        "swapstring": swapstring,
        "payment_hash": payment_hash,
        "taker_pubkey": node_info["pubkey"],
    })
    print("   ✅ Done")
    
    # Step 7: Wait for completion
    print("\n⏳ Step 7: Waiting...")
    final_status = client.wait_for_swap_completion(
        payment_hash,
        timeout_secs=180.0,
        poll_interval_secs=2.0,
    )
    print("   ✅ Swap completed!")

if __name__ == "__main__":
    atomic_swap()
```

### Real-time Quote Monitoring

```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

async function streamQuotes() {
  const client = await KaleidoClient.create({
    baseUrl: 'https://api.regtest.kaleidoswap.com',
  });

  console.log('Streaming BTC/USDT quotes...\n');

  const stream = await client.createQuoteStream('BTC/USDT');

  try {
    for (let i = 0; i < 5; i++) {
      const quoteJson = await stream.recv(10.0);

      if (quoteJson === null) {
        console.log('Timeout');
        break;
      }

      const quote = JSON.parse(quoteJson);
      console.log(`Quote ${i + 1}: Price ${quote.price}`);
    }
  } finally {
    stream.close();
  }
}

streamQuotes().catch(console.error);
```

## Error Handling

<CodeGroup>
```typescript TypeScript
import {
  KaleidoClient,
  KaleidoError,
  NetworkError,
  ValidationError,
  QuoteExpiredError,
} from '@kaleidoswap/sdk';

async function handleErrors() {
  const client = await KaleidoClient.create({
    baseUrl: 'https://api.regtest.kaleidoswap.com',
  });

  // Handle validation errors
  try {
    await client.getQuoteByPair('BTC/USDT', 100);
  } catch (error) {
    if (error instanceof ValidationError) {
      console.log(`Validation: ${error.message}`);
    } else if (error instanceof NetworkError) {
      console.log(`Network: ${error.message}`);
    } else if (error instanceof KaleidoError) {
      console.log(`SDK: ${error.message}`);
    }
  }

  // Retry on quote expiry
  async function executeWithRetry(pair: string, amount: number) {
    for (let i = 0; i < 3; i++) {
      try {
        const quote = await client.getQuoteByPair(pair, amount);
        return await client.initSwap({
          rfq_id: quote.rfqId,
          from_asset: quote.fromAsset,
          from_amount: quote.fromAmount,
          to_asset: quote.toAsset,
          to_amount: quote.toAmount,
        });
      } catch (error) {
        if (error instanceof QuoteExpiredError) {
          console.log('Quote expired, retrying...');
          continue;
        }
        throw error;
      }
    }
  }
}

handleErrors();
```

```python Python
from kaleidoswap import (
    KaleidoClient,
    KaleidoConfig,
    KaleidoError,
    NetworkError,
    ValidationError,
    QuoteExpiredError,
)

def handle_errors():
    config = KaleidoConfig(base_url="https://api.regtest.kaleidoswap.com")
    client = KaleidoClient(config)
    
    # Handle validation errors
    try:
        client.get_quote_by_pair("BTC/USDT", from_amount=100)
    except ValidationError as e:
        print(f"Validation: {e}")
    except NetworkError as e:
        print(f"Network: {e}")
    except KaleidoError as e:
        print(f"SDK: {e}")
    
    # Retry on quote expiry
    def execute_with_retry(pair, amount, max_retries=3):
        for i in range(max_retries):
            try:
                quote = client.get_quote_by_pair(pair, from_amount=amount)
                return client.init_swap({
                    "rfq_id": quote.rfq_id,
                    "from_asset": quote.from_asset.asset_id,
                    "from_amount": quote.from_asset.amount,
                    "to_asset": quote.to_asset.asset_id,
                    "to_amount": quote.to_asset.amount,
                })
            except QuoteExpiredError:
                print("Quote expired, retrying...")
                continue
        raise RuntimeError("Max retries exceeded")

if __name__ == "__main__":
    handle_errors()
```

```typescript TypeScript
      required_channel_confirmations: 3,
      funding_confirms_within_blocks: 144,
      channel_expiry_blocks: 2016,
      announce_channel: true,
```
