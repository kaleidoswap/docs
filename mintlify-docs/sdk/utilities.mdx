## Amount Conversion

The SDK provides utilities for converting between display amounts (e.g., 1.5 BTC) and atomic units (e.g., 150,000,000 satoshis).

### toRaw

Convert display amount to atomic units.

<CodeGroup>
```typescript TypeScript
const sats = await client.toRaw(0.01, 'BTC');
console.log(`0.01 BTC = ${sats} satoshis`);
// Output: 0.01 BTC = 1000000 satoshis
```

```python Python
sats = client.to_raw(0.01, "BTC")
print(f"0.01 BTC = {sats} satoshis")
# Output: 0.01 BTC = 1000000 satoshis
```
</CodeGroup>

### toDisplay

Convert atomic units to display amount.

<CodeGroup>
```typescript TypeScript
const btc = await client.toDisplay(1_000_000, 'BTC');
console.log(`1,000,000 sats = ${btc} BTC`);
// Output: 1,000,000 sats = 0.01 BTC
```

```python Python
btc = client.to_display(1_000_000, "BTC")
print(f"1,000,000 sats = {btc} BTC")
# Output: 1,000,000 sats = 0.01 BTC
```
</CodeGroup>

### convertAmount

Flexible conversion between formats.

<CodeGroup>
```typescript TypeScript
// To raw
const raw = await client.convertAmount(0.5, 'BTC', 'raw');
console.log(`0.5 BTC = ${raw} sats`);

// To display
const display = await client.convertAmount(50_000_000, 'BTC', 'display');
console.log(`50,000,000 sats = ${display} BTC`);
```

```python Python
# To raw
raw = client.convert_amount(0.5, "BTC", to="raw")
print(f"0.5 BTC = {raw} sats")

# To display
display = client.convert_amount(50_000_000, "BTC", to="display")
print(f"50,000,000 sats = {display} BTC")
```
</CodeGroup>

## Validation

### validateAmount

Validate an amount against asset constraints.

<CodeGroup>
```typescript TypeScript
const result = await client.validateAmount(0.001, 'BTC');

if (result.valid) {
  console.log(`✅ Valid: ${result.rawAmount} sats`);
} else {
  console.log(`❌ Invalid: ${result.errors.join(', ')}`);
}
```

```python Python
result = client.validate_amount(0.001, "BTC")

if result["valid"]:
    print(f"✅ Valid: {result['raw_amount']} sats")
else:
    print(f"❌ Invalid: {result['errors']}")
```
</CodeGroup>

**Returns:**
- `valid`: Whether the amount is valid
- `rawAmount` / `raw_amount`: Amount in atomic units
- `displayAmount` / `display_amount`: Amount in display units
- `errors`: List of validation errors

</CodeGroup>

## Practical Examples

### Formatting Amounts for Display

<CodeGroup>
```typescript TypeScript
async function formatBalance(
  client: KaleidoClient,
  ticker: string,
  rawBalance: number
): Promise<string> {
  const asset = await client.getAssetByTicker(ticker);
  const display = await client.toDisplay(rawBalance, asset);
  const precision = Math.min(asset.precision, 8);
  return `${display.toFixed(precision)} ${ticker}`;
}

// Usage
const formatted = await formatBalance(client, 'BTC', 150_000_000);
console.log(formatted);  // "1.50000000 BTC"
```

```python Python
def format_balance(client, ticker, raw_balance):
    asset = client.get_asset_by_ticker(ticker)
    display = client.to_display(raw_balance, asset)
    precision = min(asset.precision, 8)
    return f"{display:.{precision}f} {ticker}"

# Usage
formatted = format_balance(client, "BTC", 150_000_000)
print(formatted)  # "1.50000000 BTC"
```
</CodeGroup>

### Quote Calculator

<CodeGroup>
```typescript TypeScript
async function calculateSwap(
  client: KaleidoClient,
  fromTicker: string,
  toTicker: string,
  displayAmount: number
) {
  // Check if tradeable
  if (!await client.canTrade(fromTicker, toTicker)) {
    return { error: `Cannot trade ${fromTicker}/${toTicker}` };
  }

  // Convert to raw
  const rawAmount = await client.toRaw(displayAmount, fromTicker);

  // Get quote
  const pairTicker = `${fromTicker}/${toTicker}`;
  const quote = await client.getQuoteByPair(pairTicker, rawAmount);

  // Format output
  const outputDisplay = await client.toDisplay(quote.toAmount, toTicker);

  return {
    input: `${displayAmount} ${fromTicker}`,
    output: `${outputDisplay} ${toTicker}`,
    price: quote.price,
    fee: quote.fee.finalFee,
  };
}
```

```python Python
def calculate_swap(client, from_ticker, to_ticker, display_amount):
    # Check if tradeable
    if not client.can_trade(from_ticker, to_ticker):
        return {"error": f"Cannot trade {from_ticker}/{to_ticker}"}
    
    # Convert to raw
    raw_amount = client.to_raw(display_amount, from_ticker)
    
    # Get quote
    pair_ticker = f"{from_ticker}/{to_ticker}"
    quote = client.get_quote_by_pair(pair_ticker, from_amount=raw_amount)
    
    # Format output
    output_display = client.to_display(quote.to_asset.amount, to_ticker)
    
    return {
        "input": f"{display_amount} {from_ticker}",
        "output": f"{output_display} {to_ticker}",
        "price": quote.price,
        "fee": quote.fee.final_fee,
    }
```

### Trade Validator

<CodeGroup>
```typescript TypeScript
async function validateTrade(
  client: KaleidoClient,
  fromTicker: string,
  toTicker: string,
  amount: number
) {
  const errors: string[] = [];

  // Check pair
  if (!await client.canTrade(fromTicker, toTicker)) {
    errors.push(`Pair ${fromTicker}/${toTicker} not available`);
    return { valid: false, errors };
  }

  // Validate amount
  const validation = await client.validateAmount(amount, fromTicker);
  if (!validation.valid) {
    errors.push(...validation.errors);
  }

  // Check limits
  const pair = await client.getPairByTicker(`${fromTicker}/${toTicker}`);
  if (validation.rawAmount < pair.minBaseOrderSize) {
    errors.push(`Below minimum: ${pair.minBaseOrderSize}`);
  }
  if (validation.rawAmount > pair.maxBaseOrderSize) {
    errors.push(`Above maximum: ${pair.maxBaseOrderSize}`);
  }

  return {
    valid: errors.length === 0,
    errors,
    rawAmount: validation.rawAmount,
  };
}
```

```python Python
def validate_trade(client, from_ticker, to_ticker, amount):
    errors = []
    
    # Check pair
    if not client.can_trade(from_ticker, to_ticker):
        errors.append(f"Pair {from_ticker}/{to_ticker} not available")
        return {"valid": False, "errors": errors}
    
    # Validate amount
    validation = client.validate_amount(amount, from_ticker)
    if not validation["valid"]:
        errors.extend(validation["errors"])
    
    # Check limits
    pair = client.get_pair_by_ticker(f"{from_ticker}/{to_ticker}")
    if pair.routes:
        route = pair.routes[0]
        if route.min_from_amount and validation["raw_amount"] < route.min_from_amount:
            errors.append(f"Below minimum: {route.min_from_amount}")
        if route.max_from_amount and validation["raw_amount"] > route.max_from_amount:
            errors.append(f"Above maximum: {route.max_from_amount}")
    
    return {
        "valid": len(errors) == 0,
        "errors": errors,
        "raw_amount": validation.get("raw_amount"),
    }
```
</CodeGroup>
