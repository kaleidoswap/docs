## Exception Hierarchy

Both SDKs provide a consistent exception hierarchy:

```
KaleidoError (base)
├── APIError
│   ├── ValidationError
│   ├── AuthenticationError
│   └── RateLimitError
├── NetworkError
├── ResourceNotFoundError
│   ├── AssetNotFoundError
│   ├── TradingPairNotFoundError
│   └── OrderNotFoundError
├── QuoteExpiredError
├── InsufficientBalanceError
├── NodeNotConfiguredError
└── NodeLockedError
```

## Basic Error Handling

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, KaleidoError } from '@kaleidoswap/sdk';

const client = new KaleidoClient({
  baseUrl: 'https://api.regtest.kaleidoswap.com',
});

try {
  const assets = await client.listAssets();
  console.log(`Found ${assets.length} assets`);
} catch (error) {
  if (error instanceof KaleidoError) {
    console.log(`SDK error: ${error.message}`);
  } else {
    throw error;  // Re-throw unexpected errors
  }
}
```


## Comprehensive Error Handling

<CodeGroup>
```typescript TypeScript
import {
  KaleidoClient,
  KaleidoError,
  APIError,
  NetworkError,
  ValidationError,
  QuoteExpiredError,
  InsufficientBalanceError,
} from '@kaleidoswap/sdk';

async function getQuoteSafely(
  client: KaleidoClient,
  pair: string,
  amount: number
) {
  try {
    return await client.getQuoteByPair(pair, amount);
  } catch (error) {
    if (error instanceof ValidationError) {
      console.log(`Invalid parameters: ${error.message}`);
      return null;
    }

    if (error instanceof QuoteExpiredError) {
      console.log('Quote expired');
      return null;
    }

    if (error instanceof InsufficientBalanceError) {
      console.log(`Insufficient balance: ${error.message}`);
      return null;
    }

    if (error instanceof NetworkError) {
      console.log(`Network error: ${error.message}`);
      return null;
    }

    if (error instanceof APIError) {
      console.log(`API error: ${error.message}`);
      return null;
    }

    if (error instanceof KaleidoError) {
      console.log(`SDK error: ${error.message}`);
      return null;
    }

    throw error;
  }
}
```

## Retry Pattern

<CodeGroup>
```typescript TypeScript
import { 
  KaleidoClient, 
  NetworkError, 
  APIError, 
  RateLimitError 
} from '@kaleidoswap/sdk';

async function withRetry<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3,
  baseDelay: number = 1000
): Promise<T> {
  let lastError: Error | undefined;

  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (error instanceof RateLimitError) {
        const delay = baseDelay * Math.pow(2, attempt + 2);
        console.log(`Rate limited, waiting ${delay}ms`);
        await new Promise(r => setTimeout(r, delay));
        lastError = error;
        continue;
      }

      if (error instanceof NetworkError) {
        const delay = baseDelay * Math.pow(2, attempt);
        console.log(`Network error, retry in ${delay}ms`);
        await new Promise(r => setTimeout(r, delay));
        lastError = error;
        continue;
      }

      if (error instanceof APIError && error.message.includes('500')) {
        const delay = baseDelay * Math.pow(2, attempt);
        console.log(`Server error, retry in ${delay}ms`);
        await new Promise(r => setTimeout(r, delay));
        lastError = error;
        continue;
      }

      throw error;  // Don't retry client errors
    }
  }

  throw lastError;
}

// Usage
const assets = await withRetry(() => client.listAssets());
```

```python Python
import time
from kaleidoswap import NetworkError, APIError, RateLimitError

def with_retry(func, max_retries=3, base_delay=1.0):
    last_error = None
    
    for attempt in range(max_retries):
        try:
            return func()
            
        except RateLimitError as e:
            delay = base_delay * (2 ** (attempt + 2))
            print(f"Rate limited, waiting {delay}s")
            time.sleep(delay)
            last_error = e
            
        except NetworkError as e:
            delay = base_delay * (2 ** attempt)
            print(f"Network error, retry in {delay}s")
            time.sleep(delay)
            last_error = e
            
        except APIError as e:
            if "500" in str(e) or "server" in str(e).lower():
                delay = base_delay * (2 ** attempt)
                print(f"Server error, retry in {delay}s")
                time.sleep(delay)
                last_error = e
            else:
                raise  # Don't retry client errors
    
    if last_error:
        raise last_error

# Usage
assets = with_retry(client.list_assets)
```

## Quote Expiry Handling

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, QuoteExpiredError } from '@kaleidoswap/sdk';

async function executeSwapWithRetry(
  client: KaleidoClient,
  pair: string,
  amount: number,
  maxRetries: number = 3
) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      // Get fresh quote
      const quote = await client.getQuoteByPair(pair, amount);
      
      // Execute immediately
      return await client.initSwap({
        rfq_id: quote.rfqId,
        from_asset: quote.fromAsset,
        from_amount: quote.fromAmount,
        to_asset: quote.toAsset,
        to_amount: quote.toAmount,
      });
    } catch (error) {
      if (error instanceof QuoteExpiredError) {
        console.log(`Attempt ${i + 1}: Quote expired, retrying...`);
        continue;
      }
      throw error;
    }
  }
  throw new Error('Max retries exceeded');
}
```

```python Python

            # Get fresh quote
            quote = client.get_quote_by_pair(pair, from_amount=amount)
            
            # Execute immediately

                "rfq_id": quote.rfq_id,
                "from_asset": quote.from_asset.asset_id,
                "from_amount": quote.from_asset.amount,
                "to_asset": quote.to_asset.asset_id,
                "to_amount": quote.to_asset.amount,
            })
        except QuoteExpiredError:
            print(f"Attempt {i + 1}: Quote expired, retrying...")
            continue
    
    raise RuntimeError("Max retries exceeded")
```

```typescript TypeScript
import { KaleidoClient, NodeNotConfiguredError } from '@kaleidoswap/sdk';

async function getNodeInfo(client: KaleidoClient) {
```

## Logging Errors

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, KaleidoError, APIError, NetworkError } from '@kaleidoswap/sdk';

function loggedOperation<T>(
  name: string,
  fn: () => Promise<T>
): Promise<T> {
  console.log(`Starting: ${name}`);
  
  return fn()
    .then(result => {
      console.log(`Completed: ${name}`);
      return result;
    })
    .catch(error => {
      if (error instanceof NetworkError) {
        console.error(`Network error in ${name}: ${error.message}`);
      } else if (error instanceof APIError) {
        console.error(`API error in ${name}: ${error.message}`);
      } else if (error instanceof KaleidoError) {
        console.error(`SDK error in ${name}: ${error.message}`);
      } else {
        console.error(`Unexpected error in ${name}: ${error}`);
      }
      throw error;
    });
}

// Usage
const assets = await loggedOperation('listAssets', () => client.listAssets());
```

```python Python
import logging
from kaleidoswap import KaleidoError, APIError, NetworkError

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("kaleidoswap")

def logged_operation(name, func):
    logger.info(f"Starting: {name}")
    
    try:
        result = func()
        logger.info(f"Completed: {name}")
        return result
    except NetworkError as e:
        logger.error(f"Network error in {name}: {e}")
        raise
    except APIError as e:
        logger.error(f"API error in {name}: {e}")
        raise
    except KaleidoError as e:
        logger.error(f"SDK error in {name}: {e}")
        raise

# Usage
assets = logged_operation("list_assets", client.list_assets)
```

## Best Practices

<AccordionGroup>
  <Accordion title="Always catch KaleidoError as fallback">
    Use specific exceptions first, then fall back to the base `KaleidoError` to catch any unexpected SDK errors.
  </Accordion>
  
  <Accordion title="Check hasNode() before node operations">
    Always verify that the node URL is configured before calling any node-specific methods.
  </Accordion>
  
  <Accordion title="Handle QuoteExpiredError">
    Quotes have a limited validity period. Always be prepared to get a fresh quote if the current one expires.
  </Accordion>
  
  <Accordion title="Implement retry logic">
    For `NetworkError` and server errors (5xx), implement exponential backoff retry logic.
  </Accordion>
  
  <Accordion title="Log errors for debugging">
    Always log errors before handling them to aid in debugging and monitoring.
  </Accordion>
  
  <Accordion title="Provide user-friendly messages">
    Map SDK exceptions to user-friendly error messages appropriate for your application.
  </Accordion>
</AccordionGroup>
