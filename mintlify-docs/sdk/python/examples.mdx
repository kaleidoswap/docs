## Market Data

### List Available Assets and Trading Pairs

```python
import asyncio
from kaleidoswap_sdk import KaleidoClient

async def explore_market():
    """Explore available assets and trading pairs."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    try:
        # Get all available assets
        assets = await client.list_assets()
        print(f"📊 Found {len(assets.assets)} assets on {assets.network}\n")

        for asset in assets.assets:
            if asset.is_active:
                balance = asset.balance.spendable / (10 ** asset.precision)
                print(f"{asset.ticker:6} - {asset.name}")
                print(f"  Balance: {balance:.{asset.precision}f}")
                print(f"  Asset ID: {asset.asset_id}\n")

        # Get all trading pairs
        pairs = await client.list_pairs()
        print(f"\n💱 Active Trading Pairs:\n")

        for pair in pairs.pairs:
            if pair.is_active:
                min_order = pair.min_base_order_size / (10 ** pair.base_precision)
                print(f"{pair.base_asset}/{pair.quote_asset}")
                print(f"  Min order: {min_order} {pair.base_asset}")
                print(f"  Pair ID: {pair.id}\n")

    finally:
        await client.close()

if __name__ == "__main__":
    asyncio.run(explore_market())
```

### Compare Quotes for Different Amounts

```python
import asyncio
from kaleidoswap_sdk import KaleidoClient

asyncio.run(compare_quotes())
```

## Atomic Swaps

### Complete Atomic Swap Workflow

```python
import asyncio
from kaleidoswap_sdk import (
    KaleidoClient,
    InitMakerSwapRequest,
    ExecuteMakerSwapRequest,
    WhitelistTradeRequest,
)

async def execute_atomic_swap():
    """Execute a complete atomic swap from BTC to USDT."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    try:
        # Step 1: Get a quote
        print("Step 1: Getting quote...")
        quote = await client.get_quote(
            from_asset="BTC",
            to_asset="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
            amount=100000  # 0.001 BTC
        )
        print(f"  Quote received: {quote.exchange_rate}")

        # Step 2: Initialize the swap
        print("\nStep 2: Initializing swap...")
        swap = await client.init_maker_swap(
            InitMakerSwapRequest(
                rfq_id=quote.rfq_id,
                from_asset="BTC",
                to_asset="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
                from_amount=100000,
                to_amount=quote.to_amount
            )
        )
        print(f"  Swap initialized")
        print(f"  Payment hash: {swap.payment_hash}")

        # Step 3: Whitelist the trade on your node
        print("\nStep 3: Whitelisting trade...")
        await client.whitelist_trade(
            WhitelistTradeRequest(swapstring=swap.swapstring)
        )
        print("  Trade whitelisted")

        # Step 4: Execute the swap
        print("\nStep 4: Executing swap...")
        result = await client.execute_maker_swap(
            ExecuteMakerSwapRequest(
                payment_hash=swap.payment_hash,
                swapstring=swap.swapstring
            )
        )
        print(f"  Swap executed successfully!")

        # Step 5: Wait for completion
        print("\nStep 5: Waiting for swap completion...")
        final_status = await client.wait_for_swap_completion(
            payment_hash=swap.payment_hash,
            timeout_seconds=300,
            poll_interval=5
        )
        print(f"  Swap completed with status: {final_status.status}")

    except Exception as e:
        print(f"Error during swap: {e}")
    finally:
        await client.close()

asyncio.run(execute_atomic_swap())
```

## Order Management

### Create and Track an Order

```python
import asyncio
from kaleidoswap_sdk import (
    KaleidoClient,
    CreateOrderRequest,
    GetOrderRequest,
    SwapOrderStatusRequest,
)

async def create_and_track_order():
    """Create an order and track its status."""

    async with KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    ) as client:
        # Get a quote first
        quote = await client.get_quote(
            from_asset="BTC",
            to_asset="USDT",
            amount=100000
        )

        # Create the order
        print("Creating order...")
        order = await client.create_order(
            CreateOrderRequest(
                rfq_id=quote.rfq_id,
                from_type="ONCHAIN",
                to_type="LIGHTNING",
                from_asset="BTC",
                to_asset="USDT",
                from_amount=100000,
                to_amount=quote.to_amount
            )
        )

        print(f"\n✅ Order Created!")
        print(f"  Order ID: {order.id}")
        print(f"  Status: {order.status}")

        if order.pay_in == "ONCHAIN":
            print(f"  Payment Address: {order.onchain_address}")
            print(f"\n  Send BTC to the address above to complete the order")
        elif order.pay_in == "LIGHTNING":
            print(f"  Lightning Invoice: {order.ln_invoice}")
            print(f"\n  Pay the invoice above to complete the order")

        # Poll order status
        print("\n  Monitoring order status...")
        while True:
            await asyncio.sleep(5)

            status = await client.get_swap_order_status(
                SwapOrderStatusRequest(order_id=order.id)
            )

            print(f"  Status: {status.status}")

            if status.status in ["COMPLETED", "FAILED", "CANCELLED"]:
                break

        print(f"\n  Order final status: {status.status}")

asyncio.run(create_and_track_order())
```

## Lightning Operations

### Open a Lightning Channel

```python
import asyncio
from kaleidoswap_sdk import KaleidoClient, OpenChannelRequest, ConnectPeerRequest

async def open_channel_with_lsp():
    """Connect to LSP and open a channel."""

    async with KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    ) as client:
        # Get LSP connection info
        lsp_url = await client.get_lsp_connection_url()
        print(f"LSP Connection URL: {lsp_url}")

        # Connect to LSP peer
        print("\nConnecting to LSP...")
        await client.connect_peer(
            ConnectPeerRequest(connection_url=lsp_url)
        )
        print("Connected to LSP")

        # Get LSP info
        lsp_info = await client.get_lsp_info()

        # Open a channel
        print("\nOpening channel...")
        channel = await client.open_channel(
            OpenChannelRequest(
                peer_pubkey=lsp_info.pubkey,
                capacity_sat=1000000,  # 1M sats
                push_msat=0,
                asset_amount=100000000,  # 100 USDT
                asset_id="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
                public=True,
                with_anchors=True
            )
        )

        print(f"✅ Channel opened!")
        print(f"  Temporary Channel ID: {channel.temporary_channel_id}")

asyncio.run(open_channel_with_lsp())
```

### Create and Monitor Invoice

```python
import asyncio
from kaleidoswap_sdk import (
    KaleidoClient,
    LNInvoiceRequest,
    InvoiceStatusRequest,
    InvoiceStatus,
)

async def create_and_monitor_invoice():
    """Create an invoice and monitor for payment."""

    async with KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    ) as client:
        # Create invoice
        invoice = await client.generate_ln_invoice(
            LNInvoiceRequest(
                amount_msat=1000000,  # 1000 sats
                description="Payment for services",
                expiry_secs=3600  # 1 hour
            )
        )

        print(f"📄 Invoice Created!")
        print(f"  Invoice: {invoice.invoice}")
        print(f"  Payment Hash: {invoice.payment_hash}")
        print(f"\n  Waiting for payment...")

        # Monitor invoice status
        while True:
            await asyncio.sleep(5)

            status = await client.get_invoice_status(
                InvoiceStatusRequest(payment_hash=invoice.payment_hash)
            )

            if status.status == InvoiceStatus.Succeeded:
                print(f"\n  ✅ Payment received!")
                break
            elif status.status == InvoiceStatus.Failed:
                print(f"\n  ❌ Payment failed")
                break
            elif status.status == InvoiceStatus.Expired:
                print(f"\n  ⏰ Invoice expired")
                break

asyncio.run(create_and_monitor_invoice())
```

## Real-time Data with WebSocket

### Subscribe to Live Quote Updates

```python
import asyncio
from kaleidoswap_sdk import KaleidoClient

async def stream_live_quotes():
    """Subscribe to real-time quote updates."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    # Callback for quote updates
    async def handle_quote(quote):
        btc_amount = 0.001  # 0.001 BTC
        usdt_amount = quote.to_amount / 1_000_000
        print(f"📊 Live Quote: {btc_amount} BTC → {usdt_amount:.2f} USDT")
        print(f"   Rate: {quote.exchange_rate:.2f}")
        print(f"   Time: {quote.timestamp}\n")

    try:
        # Connect WebSocket
        await client.connect()
        print("🔌 Connected to WebSocket\n")

        # Subscribe to BTC/USDT quotes
        await client.get_quote_websocket(
            from_asset="BTC",
            to_asset="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
            amount=100000,  # 0.001 BTC
            callback=handle_quote
        )

        # Keep receiving updates for 60 seconds
        await asyncio.sleep(60)

    finally:
        await client.disconnect()
        await client.close()
        print("Disconnected")

asyncio.run(stream_live_quotes())
```

## Error Handling

### Robust Error Handling Pattern

```python
import asyncio
from kaleidoswap_sdk import KaleidoClient
from kaleidoswap_sdk.exceptions import (
    KaleidoException,
    NetworkError,
    ValidationError,
    TimeoutError,
    SwapError,
)

async def robust_swap_with_error_handling():
    """Execute a swap with comprehensive error handling."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    try:
        # Get quote
        quote = await client.get_quote(
            from_asset="BTC",
            to_asset="USDT",
            amount=100000
        )
        print(f"Quote received: {quote.rfq_id}")

    except NetworkError as e:
        print(f"❌ Network error: {e}")
        print("   Check your internet connection")
    except ValidationError as e:
        print(f"❌ Validation error: {e}")
        print("   Check your input parameters")
    except TimeoutError as e:
        print(f"❌ Timeout error: {e}")
        print("   Request took too long, try again")
    except SwapError as e:
        print(f"❌ Swap error: {e}")
        print("   The swap could not be completed")
    except KaleidoException as e:
        print(f"❌ API error: {e}")
    except Exception as e:
        print(f"❌ Unexpected error: {e}")
    finally:
        await client.close()

asyncio.run(robust_swap_with_error_handling())
```

## Best Practices

### Using Context Managers

Always use async context managers for automatic cleanup:

```python
async with KaleidoClient(api_url="...", node_url="...") as client:
    # Your code here
    # Client automatically closes when done
    pass
```

### Environment Variables

Store sensitive configuration in environment variables:

```python
import os
from kaleidoswap_sdk import KaleidoClient

API_URL = os.getenv("KALEIDO_API_URL", "https://api.staging.kaleidoswap.com/api/v1")
NODE_URL = os.getenv("KALEIDO_NODE_URL")

async with KaleidoClient(api_url=API_URL, node_url=NODE_URL) as client:
    # Your code
    pass
```

### Logging

Enable logging for debugging:

```python
import logging

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s"
)
```
