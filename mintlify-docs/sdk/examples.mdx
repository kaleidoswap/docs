## Market Data

### List Available Assets and Trading Pairs

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/kaleidoswap-sdk';

async function exploreMarket() {
  const client = new KaleidoClient({
    baseUrl: 'https://api.staging.kaleidoswap.com/api/v1'
  });

  try {
    // Get all available assets
    const assets = await client.listAssets();
    console.log(`📊 Found ${assets.assets.length} assets on ${assets.network}\n`);

    for (const asset of assets.assets) {
      if (asset.is_active) {
        const balance = asset.balance.spendable / (10 ** asset.precision);
        console.log(`${asset.ticker.padEnd(6)} - ${asset.name}`);
        console.log(`  Balance: ${balance.toFixed(asset.precision)}`);
        console.log(`  Asset ID: ${asset.asset_id}\n`);
      }
    }

    // Get all trading pairs
    const pairs = await client.listPairs();
    console.log(`\n💱 Active Trading Pairs:\n`);

    for (const pair of pairs.pairs) {
      if (pair.is_active) {
        const minOrder = pair.min_base_order_size / (10 ** pair.base_precision);
        console.log(`${pair.base_asset}/${pair.quote_asset}`);
        console.log(`  Min order: ${minOrder} ${pair.base_asset}`);
        console.log(`  Pair ID: ${pair.id}\n`);
      }
    }
  } catch (error) {
    console.error('Error:', error);
  }
}

exploreMarket();
```

```python Python
import asyncio
from kaleidoswap_sdk import KaleidoClient

async def explore_market():
    """Explore available assets and trading pairs."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    try:
        # Get all available assets
        assets = await client.list_assets()
        print(f"📊 Found {len(assets.assets)} assets on {assets.network}\n")

        for asset in assets.assets:
            if asset.is_active:
                balance = asset.balance.spendable / (10 ** asset.precision)
                print(f"{asset.ticker:6} - {asset.name}")
                print(f"  Balance: {balance:.{asset.precision}f}")
                print(f"  Asset ID: {asset.asset_id}\n")

        # Get all trading pairs
        pairs = await client.list_pairs()
        print(f"\n💱 Active Trading Pairs:\n")

        for pair in pairs.pairs:
            if pair.is_active:
                min_order = pair.min_base_order_size / (10 ** pair.base_precision)
                print(f"{pair.base_asset}/{pair.quote_asset}")
                print(f"  Min order: {min_order} {pair.base_asset}")
                print(f"  Pair ID: {pair.id}\n")

    finally:
        await client.close()

if __name__ == "__main__":
    asyncio.run(explore_market())
```

## Atomic Swaps

### Complete Atomic Swap Workflow

For more control, you can execute each step of the swap process manually.

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/kaleidoswap-sdk';

async function executeAtomicSwap() {
  const client = new KaleidoClient({
    baseUrl: 'https://api.staging.kaleidoswap.com/api/v1',
    nodeUrl: '<Your node URL>'
  });

  try {
    // Step 1: Get a quote
    console.log('Step 1: Getting quote...');
    const quote = await client.quoteRequest(
      'BTC',
      'rgb:im7mWgoS-4QX_1b1-DT23iPq-niObDFM-qGN7R2x-lFLJYak',
      1000000 // 0.01 BTC (minimum)
    );
    console.log(`  Quote received: ${quote.exchange_rate}`);

    // Step 2: Initialize the swap
    console.log('\nStep 2: Initializing swap...');
    const swap = await client.initMakerSwap({
      rfq_id: quote.rfq_id,
      from_asset: 'BTC',
      to_asset: 'rgb:im7mWgoS-4QX_1b1-DT23iPq-niObDFM-qGN7R2x-lFLJYak',
      from_amount: 1000000,
      to_amount: quote.to_amount
    });
    console.log('  Swap initialized');
    console.log(`  Payment hash: ${swap.payment_hash}`);

    // Step 3: Whitelist the trade on your node
    console.log('\nStep 3: Whitelisting trade...');
    await client.whitelistTrade({ swapstring: swap.swapstring });
    console.log('  Trade whitelisted');

    // Step 4: Execute the swap
    console.log('\nStep 4: Executing swap...');
    const result = await client.executeMakerSwap({
      payment_hash: swap.payment_hash,
      swapstring: swap.swapstring
    });
    console.log('  Swap executed successfully!');

    // Step 5: Wait for completion
    console.log('\nStep 5: Waiting for swap completion...');
    const finalStatus = await client.waitForSwapCompletion(
      swap.payment_hash,
      300, // timeout_seconds
      5 // poll_interval
    );
    console.log(`  Swap completed with status: ${finalStatus.status}`);

  } catch (error) {
    console.error('Error during swap:', error);
  }
}

executeAtomicSwap();
```

```python Python
import asyncio
from kaleidoswap_sdk import (
    KaleidoClient,
    InitMakerSwapRequest,
    ExecuteMakerSwapRequest,
    WhitelistTradeRequest,
)

async def execute_atomic_swap():
    """Execute a complete atomic swap from BTC to USDT."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    try:
        # Step 1: Get a quote
        print("Step 1: Getting quote...")
        quote = await client.get_quote(
            from_asset="BTC",
            to_asset="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
            amount=100000  # 0.001 BTC
        )
        print(f"  Quote received: {quote.exchange_rate}")

        # Step 2: Initialize the swap
        print("\nStep 2: Initializing swap...")
        swap = await client.init_maker_swap(
            InitMakerSwapRequest(
                rfq_id=quote.rfq_id,
                from_asset="BTC",
                to_asset="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
                from_amount=100000,
                to_amount=quote.to_amount
            )
        )
        print(f"  Swap initialized")
        print(f"  Payment hash: {swap.payment_hash}")

        # Step 3: Whitelist the trade on your node
        print("\nStep 3: Whitelisting trade...")
        await client.whitelist_trade(
            WhitelistTradeRequest(swapstring=swap.swapstring)
        )
        print("  Trade whitelisted")

        # Step 4: Execute the swap
        print("\nStep 4: Executing swap...")
        result = await client.execute_maker_swap(
            ExecuteMakerSwapRequest(
                payment_hash=swap.payment_hash,
                swapstring=swap.swapstring
            )
        )
        print(f"  Swap executed successfully!")

        # Step 5: Wait for completion
        print("\nStep 5: Waiting for swap completion...")
        final_status = await client.wait_for_swap_completion(
            payment_hash=swap.payment_hash,
            timeout_seconds=300,
            poll_interval=5
        )
        print(f"  Swap completed with status: {final_status.status}")

    except Exception as e:
        print(f"Error during swap: {e}")
    finally:
        await client.close()

asyncio.run(execute_atomic_swap())
```

## Order Management

### Create and Track an Order

This example shows the complete flow of creating an order and monitoring its status until completion.

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/kaleidoswap-sdk';

async function createAndTrackOrder() {
  const client = new KaleidoClient({
    baseUrl: 'https://api.staging.kaleidoswap.com/api/v1'
  });

  try {
    // Step 1: Get a quote
    console.log('Step 1: Getting quote...');
    const quote = await client.getQuoteByPair('BTC/USDT', { fromAmount: 100000 });
    
    // Create the order
    console.log('Creating order...');
    const order = await client.createOrder({
      rfq_id: quote.rfq_id,
      from_type: 'ONCHAIN',
      to_type: 'LIGHTNING',
      from_asset: quote.from_asset,
      to_asset: quote.to_asset,
      from_amount: 100000,
      to_amount: quote.to_amount
    });

    console.log('\n✅ Order Created!');
    console.log(`  Order ID: ${order.id}`);
    console.log(`  Status: ${order.status}`);

    if (order.pay_in === 'ONCHAIN') {
      console.log(`  Payment Address: ${order.onchain_address}`);
      console.log('\n  Send BTC to the address above to complete the order');
    } else if (order.pay_in === 'LIGHTNING') {
      console.log(`  Lightning Invoice: ${order.ln_invoice}`);
      console.log('\n  Pay the invoice above to complete the order');
    }

    // Poll order status
    console.log('\n  Monitoring order status...');
    while (true) {
      await new Promise(resolve => setTimeout(resolve, 5000));

      const status = await client.swapOrderStatus(order.id);
      console.log(`  Status: ${status.status}`);

      if (['COMPLETED', 'FAILED', 'CANCELLED'].includes(status.status)) {
        break;
      }
    }

    console.log(`\n  Order final status: ${status.status}`);

  } catch (error) {
    console.error('Error:', error);
  }
}

createAndTrackOrder();
```

```python Python
import asyncio
from kaleidoswap_sdk import (
    KaleidoClient,
    CreateOrderRequest,
    SwapOrderStatusRequest,
)

async def create_and_track_order():
    """Create an order and track its status."""

    async with KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    ) as client:
        # Get a quote first
        quote = await client.get_quote(
            from_asset="BTC",
            to_asset="rgb:im7mWgoS-4QX_1b1-DT23iPq-niObDFM-qGN7R2x-lFLJYak",
            amount=1000000
        )

        # Create the order
        print("Creating order...")
        order = await client.create_order(
            CreateOrderRequest(
                rfq_id=quote.rfq_id,
                from_type="ONCHAIN",
                to_type="LIGHTNING",
                from_asset="BTC",
                to_asset="rgb:im7mWgoS-4QX_1b1-DT23iPq-niObDFM-qGN7R2x-lFLJYak",
                from_amount=1000000,
                to_amount=quote.to_amount
            )
        )

        print(f"\n✅ Order Created!")
        print(f"  Order ID: {order.id}")
        print(f"  Status: {order.status}")

        if order.pay_in == "ONCHAIN":
            print(f"  Payment Address: {order.onchain_address}")
            print(f"\n  Send BTC to the address above to complete the order")
        elif order.pay_in == "LIGHTNING":
            print(f"  Lightning Invoice: {order.ln_invoice}")
            print(f"\n  Pay the invoice above to complete the order")

        # Poll order status
        print("\n  Monitoring order status...")
        while True:
            await asyncio.sleep(5)

            status = await client.get_swap_order_status(
                SwapOrderStatusRequest(order_id=order.id)
            )

            print(f"  Status: {status.status}")

            if status.status in ["COMPLETED", "FAILED", "CANCELLED"]:
                break

        print(f"\n  Order final status: {status.status}")

asyncio.run(create_and_track_order())
```

```python Python
                amount_msat=1000000,  # 1000 sats
                description="Payment for services",
                expiry_secs=3600  # 1 hour
            )
        )

        print(f"📄 Invoice Created!")
        print(f"  Invoice: {invoice.invoice}")
        print(f"  Payment Hash: {invoice.payment_hash}")
        print(f"\n  Waiting for payment...")

        # Monitor invoice status
        while True:
            await asyncio.sleep(5)

            status = await client.get_invoice_status(
                InvoiceStatusRequest(payment_hash=invoice.payment_hash)
            )

            if status.status == InvoiceStatus.Succeeded:
                print(f"\n  ✅ Payment received!")
                break
            elif status.status == InvoiceStatus.Failed:
                print(f"\n  ❌ Payment failed")
                break
            elif status.status == InvoiceStatus.Expired:
                print(f"\n  ⏰ Invoice expired")
                break

asyncio.run(create_and_monitor_invoice())
```

## Real-time Data with WebSocket

### Subscribe to Live Quote Updates

<Note>
This example is Python-specific. WebSocket support for TypeScript is coming soon.
</Note>

```python
import asyncio
from kaleidoswap_sdk import KaleidoClient

async def stream_live_quotes():
    """Subscribe to real-time quote updates."""

    client = KaleidoClient(
        api_url="https://api.staging.kaleidoswap.com/api/v1",
        node_url="<Your node URL>"
    )

    # Callback for quote updates
    async def handle_quote(quote):
        btc_amount = 0.001  # 0.001 BTC
        usdt_amount = quote.to_amount / 1_000_000
        print(f"📊 Live Quote: {btc_amount} BTC → {usdt_amount:.2f} USDT")
        print(f"   Rate: {quote.exchange_rate:.2f}")
        print(f"   Time: {quote.timestamp}\n")

    try:
        # Connect WebSocket
        await client.connect()
        print("🔌 Connected to WebSocket\n")

        # Subscribe to BTC/USDT quotes
        await client.get_quote_websocket(
            from_asset="BTC",
            to_asset="rgb:q1O5Mn5y-7EoxdTy-xu3ChkP-HmhgGvJ-vQ3ryQ9-CcMxkfg",
            amount=100000,  # 0.001 BTC
            callback=handle_quote
        )

        # Keep receiving updates for 60 seconds
        await asyncio.sleep(60)

    finally:
        await client.disconnect()
        await client.close()
        print("Disconnected")

asyncio.run(stream_live_quotes())
```
