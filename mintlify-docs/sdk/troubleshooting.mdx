#### Validate Before Trading

<CodeGroup>

```typescript
function validateOrderSize(amount: number, ticker: string) {
  const asset = assetMapper.findByTicker(ticker);
  if (!asset) {
    throw new Error(`Asset ${ticker} not found`);
  }

  const validation = precisionHandler.validateOrderSize(amount, asset);
  if (!validation.valid) {
    const limits = precisionHandler.getOrderSizeLimits(asset);
    console.error(`Invalid order size: ${validation.error}`);
    console.log(`Valid range: ${limits.minDecimal} - ${limits.maxDecimal} ${ticker}`);
    return false;
  }

  return true;
}

// Usage
if (validateOrderSize(0.001, 'BTC')) {
  // Proceed with order
}
```

```python
def validate_order_size(amount: float, ticker: str) -> bool:

        print(f"Invalid order size: {validation.error}")
        print(
            f"Valid range: {limits['min_decimal']} - {limits['max_decimal']} {ticker}"
```

### 5. Rate Limiting Errors

**Problem:** Hitting API rate limits with 429 errors.

**Symptoms:**

```
Error: Rate limit exceeded
HTTP 429 Too Many Requests
```

**Solutions:**

#### Implement Client-Side Rate Limiting

<CodeGroup>

```typescript
class RateLimitedClient {
  private lastRequest = 0;
  private minInterval = 1000; // 1 second between requests

  async makeRequest<T>(requestFn: () => Promise<T>): Promise<T> {
    const now = Date.now();
    const timeSinceLastRequest = now - this.lastRequest;

    if (timeSinceLastRequest < this.minInterval) {
      const waitTime = this.minInterval - timeSinceLastRequest;
      console.log(`Rate limiting: waiting ${waitTime}ms`);
      await new Promise(resolve => setTimeout(resolve, waitTime));
    }

    this.lastRequest = Date.now();
    return await requestFn();
  }
}

// Usage
const rateLimitedClient = new RateLimitedClient();

const quote = await rateLimitedClient.makeRequest(() =>
  client.quoteRequest(from, to, amount)
);
```

```python
import asyncio
from datetime import datetime

class RateLimitedClient:
    def __init__(self, min_interval_seconds: float = 1):
        self.last_request = 0
        self.min_interval = min_interval_seconds

    async def make_request(self, request_fn):
        now = datetime.now().timestamp()
        time_since_last_request = now - self.last_request

        if time_since_last_request < self.min_interval:
            wait_time = self.min_interval - time_since_last_request
            print(f'Rate limiting: waiting {wait_time:.2f}s')
            await asyncio.sleep(wait_time)

        self.last_request = datetime.now().timestamp()
        return await request_fn()

# Usage
rate_limited_client = RateLimitedClient(min_interval_seconds=1)

    lambda: client.quote_request(from_id, to_id, amount)
)
```

### 6. Network Timeout Issues

**Problem:** Requests timeout or fail due to network issues.

**Symptoms:**

```
Error: Network timeout
Error: Connection refused
Error: DNS resolution failed
```

**Solutions:**

#### Increase Timeout Values

<CodeGroup>

```typescript
async function requestWithTimeout<T>(
  requestFn: () => Promise<T>,
  timeoutMs: number = 30000
): Promise<T> {
  return Promise.race([
    requestFn(),
    new Promise<never>((_, reject) =>
      setTimeout(() => reject(new Error('Request timeout')), timeoutMs)
    )
  ]);
}

// Usage
const quote = await requestWithTimeout(
  () => client.quoteRequest(from, to, amount),
  30000 // 30 seconds
);
```

```python
import asyncio

async def request_with_timeout(request_fn, timeout_seconds: float = 30):
    try:
        return await asyncio.wait_for(request_fn(), timeout=timeout_seconds)
    except asyncio.TimeoutError:
        raise TimeoutError(f'Request timed out after {timeout_seconds} seconds')

# Usage
quote = await request_with_timeout(
    lambda: client.quote_request(from_id, to_id, amount),
    timeout_seconds=30
```

#### Implement Retry with Exponential Backoff

<CodeGroup>

```typescript
async function retryWithBackoff<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3
): Promise<T> {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (attempt === maxRetries) {
        throw error;
      }

      const delay = Math.min(1000 * Math.pow(2, attempt - 1), 10000);
      console.log(`Attempt ${attempt} failed, retrying in ${delay}ms...`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }

  throw new Error('Max retries exceeded');
}
```

```python
import asyncio
import random

async def retry_with_backoff(fn, max_retries: int = 3):
    for attempt in range(1, max_retries + 1):
        try:
            return await fn()
        except Exception as error:
            if attempt == max_retries:
                raise error

            delay = min(1 * (2 ** (attempt - 1)), 10)
            # Add jitter
            delay = delay + random.uniform(0, delay * 0.1)
            print(f'Attempt {attempt} failed, retrying in {delay:.2f}s...')
            await asyncio.sleep(delay)

    raise RuntimeError('Max retries exceeded')
```

### 7. Precision and Rounding Errors

**Problem:** Incorrect amount calculations due to floating-point precision.

**Symptoms:**

```

Actual: 99999999 satoshis
```

**Solutions:**

#### Always Use PrecisionHandler

#### Validate Conversions

<CodeGroup>

```typescript
function validateConversion(decimalAmount: number, assetId: string) {
  const atomicAmount = precisionHandler.toAtomicAmount(decimalAmount, assetId);
  const backToDecimal = precisionHandler.toDecimalAmount(atomicAmount, assetId);

  const difference = Math.abs(decimalAmount - backToDecimal);
  const tolerance = Math.pow(10, -precisionHandler.getPrecision(assetId));

  if (difference > tolerance) {
    console.warn(`Precision loss detected: ${decimalAmount} -> ${backToDecimal}`);
  }

  return atomicAmount;
}
```

```python
def validate_conversion(decimal_amount: float, asset_id: str) -> int:
    atomic_amount = precision_handler.to_atomic_amount(decimal_amount, asset_id)
    back_to_decimal = precision_handler.to_decimal_amount(atomic_amount, asset_id)

    difference = abs(decimal_amount - back_to_decimal)
    precision = precision_handler.get_precision(asset_id)
    tolerance = 10 ** (-precision)

    if difference > tolerance:
        logger.warning(
            f'Precision loss detected: {decimal_amount} -> {back_to_decimal}'
        )

    return atomic_amount
```

```python
import logging
from aiohttp import ClientSession

class DebuggingClient:
    def __init__(self, base_url: str, **kwargs):
        self.base_url = base_url
        self.logger = logging.getLogger(__name__)

    async def get(self, endpoint: str):
        url = f'{self.base_url}{endpoint}'
        self.logger.debug(f'GET {url}')
        start = datetime.now()

        try:
            async with ClientSession() as session:
                async with session.get(url) as response:
                    result = await response.json()
            duration = (datetime.now() - start).total_seconds()
            self.logger.debug(f'GET {url} - {duration:.2f}s - Success')
            return result
        except Exception as error:
            duration = (datetime.now() - start).total_seconds()
            self.logger.debug(f'GET {url} - {duration:.2f}s - Error: {error}')
            raise

    async def post(self, endpoint: str, data: dict):
```
