## Configuration

### Basic Setup

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

const client = await KaleidoClient.create({
  baseUrl: 'https://api.regtest.kaleidoswap.com',
});

// Make your first API call
const assets = await client.listAssets();
console.log(`Found ${assets.length} assets`);
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig

config = KaleidoConfig(base_url="https://api.regtest.kaleidoswap.com")
client = KaleidoClient(config)

# Make your first API call
assets = client.list_assets()
print(f"Found {len(assets)} assets")
```
</CodeGroup>

### Full Configuration

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, KaleidoConfig } from '@kaleidoswap/sdk';

const config: KaleidoConfig = {
  // Required: API base URL
  baseUrl: 'https://api.regtest.kaleidoswap.com',

  // Optional: Your RGB Lightning Node URL (for swap execution)
  nodeUrl: 'http://localhost:3001',

  // Optional: API key for authenticated requests
  apiKey: undefined,

  // Optional: Request timeout in seconds (default: 30.0)
  timeout: 30.0,

  // Optional: Maximum retries (default: 3)
  maxRetries: 3,
};

const client = await KaleidoClient.create(config);
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig

config = KaleidoConfig(
    # Required: API base URL
    base_url="https://api.regtest.kaleidoswap.com",

    # Optional: Your RGB Lightning Node URL (for swap execution)
    node_url="http://localhost:3001",

    # Optional: API key for authenticated requests
    api_key=None,

    # Optional: Request timeout in seconds (default: 30.0)
    timeout=30.0,

    # Optional: Maximum retries (default: 3)
    max_retries=3,
)

client = KaleidoClient(config)
```
</CodeGroup>

### Environment Variables

You can also configure the SDK using environment variables:

```bash
export KALEIDO_API_URL=https://api.regtest.kaleidoswap.com
export KALEIDO_NODE_URL=http://localhost:3001
export KALEIDO_API_KEY=your_api_key_here
```

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

const client = await KaleidoClient.create({
  baseUrl: process.env.KALEIDO_API_URL || 'https://api.regtest.kaleidoswap.com',
  nodeUrl: process.env.KALEIDO_NODE_URL || null,
  apiKey: process.env.KALEIDO_API_KEY || null,
});
```

```python Python
import os
from kaleidoswap import KaleidoClient, KaleidoConfig

config = KaleidoConfig(
    base_url=os.environ.get("KALEIDO_API_URL", "https://api.regtest.kaleidoswap.com"),
    node_url=os.environ.get("KALEIDO_NODE_URL"),
    api_key=os.environ.get("KALEIDO_API_KEY"),
)

client = KaleidoClient(config)
```
</CodeGroup>

## Sub-Client Architecture

The SDK organizes API operations into domain-specific sub-clients:

<CodeGroup>
```typescript TypeScript
// Market operations - assets, pairs, quotes
const assets = await client.listAssets();
const pairs = await client.listPairs();
const quote = await client.getQuoteByPair('BTC/USDT', 1_000_000n);

// Order management
const history = await client.getOrderHistory();
const status = await client.getSwapOrderStatus('order-id');

// Swap operations
const result = await client.createSwapOrder(request);

// LSP operations
const lspInfo = await client.getLspInfo();

// RGB Node operations (requires nodeUrl)
if (client.hasNode()) {
  const nodeInfo = await client.getNodeInfo();
}
```

```python Python
# Market operations - assets, pairs, quotes
assets = client.market.list_assets()
pairs = client.market.list_pairs()
quote = client.market.get_quote_by_pair("BTC/USDT", from_amount=1_000_000)

# Order management
history = client.orders.get_order_history()
status = client.orders.get_swap_order_status("order-id")

# Swap operations
result = client.swaps.init_swap(request)

# LSP operations
lsp_info = client.lsp.get_lsp_info()

# RGB Node operations (requires node_url)
if client.has_node():
    channels = client.node.list_channels()
    balance = client.node.get_btc_balance()
```
</CodeGroup>

## Checking Node Configuration

Before executing swaps or node operations, verify that your node is configured:

<CodeGroup>
```typescript TypeScript
if (client.hasNode()) {
  console.log('✅ Node configured - can execute swaps');
  
  const nodeInfo = await client.getRgbNodeInfo();
  console.log(`Node pubkey: ${nodeInfo.pubkey}`);
} else {
  console.log('⚠️ Node not configured - limited to market data');
}
```


## Basic Error Handling

<CodeGroup>
```typescript TypeScript
import { 
  KaleidoClient, 
  KaleidoError,
  NetworkError,
  ValidationError,
  APIError,
} from '@kaleidoswap/sdk';

try {
  const pairs = await client.listPairs();
  console.log(`Success: ${pairs.length} pairs found`);
} catch (error) {
  if (error instanceof NetworkError) {
    console.log(`Network error: ${error.message}`);
  } else if (error instanceof ValidationError) {
    console.log(`Validation error: ${error.message}`);
  } else if (error instanceof APIError) {
    console.log(`API error: ${error.message}`);
  } else if (error instanceof KaleidoError) {
    console.log(`SDK error: ${error.message}`);
  }
}
```

```python Python
from kaleidoswap import (
    KaleidoClient,
    KaleidoConfig,
    KaleidoError,
    NetworkError,
    ValidationError,
    APIError,
)

try:
    pairs = client.list_pairs()
    print(f"Success: {len(pairs)} pairs found")
except NetworkError as e:
    print(f"Network error: {e}")
except ValidationError as e:
    print(f"Validation error: {e}")
except APIError as e:
    print(f"API error: {e}")
except KaleidoError as e:
    print(f"SDK error: {e}")
```
</CodeGroup>

## Complete Example

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, KaleidoError } from '@kaleidoswap/sdk';

async function getMarketOverview() {
  const client = await KaleidoClient.create({
    baseUrl: process.env.KALEIDO_API_URL || 'https://api.regtest.kaleidoswap.com',
  });

  try {
    // Fetch market data
    console.log('Fetching market data...\n');

    const [assets, pairs] = await Promise.all([
      client.listAssets(),
      client.listPairs(),
    ]);

    console.log('📊 Market Overview');
    console.log(`   Assets: ${assets.length}`);
    console.log(`   Trading Pairs: ${pairs.length}`);

    // Display trading pairs
    console.log('\n💱 Active Trading Pairs:');
    pairs.forEach(pair => {
      // Assuming pair has isActive property, adjust based on actual type
      console.log(`   ${pair.baseAsset}/${pair.quoteAsset}`);
    });

    // Get a sample quote
    if (pairs.length > 0) {
      const pair = pairs[0];
      const pairTicker = `${pair.baseAsset}/${pair.quoteAsset}`;
      // Amounts should be BigInt or number. Using BigInt literal 1_000_000n
      const amount = 1_000_000n;
      
      const quote = await client.getQuoteByPair(pairTicker, amount);
      
      console.log(`\n💰 Sample Quote for ${pairTicker}:`);
      console.log(`   From: ${quote.fromAmount}`);
      console.log(`   To: ${quote.toAmount}`);
      console.log(`   Price: ${quote.price}`);
    }

  } catch (error) {
    if (error instanceof KaleidoError) {
      console.error(`Error: ${error.message}`);
    } else {
      throw error;
    }
  }
}

getMarketOverview();
```

```python Python
from kaleidoswap import KaleidoClient, KaleidoConfig, KaleidoError
import os

def get_market_overview():
    config = KaleidoConfig(
        base_url=os.environ.get("KALEIDO_API_URL", "https://api.regtest.kaleidoswap.com")
    )
    client = KaleidoClient(config)

    try:
        # Fetch market data
        print("Fetching market data...\n")

        assets = client.list_assets()
        pairs = client.list_pairs()

        print("📊 Market Overview")
        print(f"   Assets: {len(assets)}")
        print(f"   Trading Pairs: {len(pairs)}")

        # Display trading pairs
        print("\n💱 Active Trading Pairs:")
        for pair in pairs:
            if pair.is_active:
                base = pair.base.ticker if pair.base else "?"
                quote = pair.quote.ticker if pair.quote else "?"
                print(f"   {base}/{quote}")

        # Get a sample quote
        if pairs:
            pair = pairs[0]
            base = pair.base.ticker if pair.base else "BTC"
            quote = pair.quote.ticker if pair.quote else "USDT"
            pair_ticker = f"{base}/{quote}"
            
            # Use minimum amount from routes if available
            amount = 1_000_000
            if pair.routes and pair.routes[0].min_from_amount:
                amount = pair.routes[0].min_from_amount
            
            swap_quote = client.get_quote_by_pair(pair_ticker, from_amount=amount)
            
            print(f"\n💰 Sample Quote for {pair_ticker}:")
            print(f"   From: {swap_quote.from_asset.amount}")
            print(f"   To: {swap_quote.to_asset.amount}")
            print(f"   Price: {swap_quote.price}")

    except KaleidoError as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    get_market_overview()
```
