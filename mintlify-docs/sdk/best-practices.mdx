## Client Initialization

### Use Environment Variables

<CodeGroup>
```typescript TypeScript
import { KaleidoClient } from '@kaleidoswap/sdk';

const client = new KaleidoClient({
  baseUrl: process.env.KALEIDO_API_URL || 'https://api.regtest.kaleidoswap.com',
  nodeUrl: process.env.KALEIDO_NODE_URL,
  apiKey: process.env.KALEIDO_API_KEY,
});
```

```python Python
import os
from kaleidoswap import KaleidoClient, KaleidoConfig

config = KaleidoConfig(
    base_url=os.environ.get("KALEIDO_API_URL", "https://api.regtest.kaleidoswap.com"),
    node_url=os.environ.get("KALEIDO_NODE_URL"),
    api_key=os.environ.get("KALEIDO_API_KEY"),
)
client = KaleidoClient(config)
```
</CodeGroup>

### Create a Singleton Client

For most applications, reuse a single client instance:

<CodeGroup>
```typescript TypeScript
// lib/kaleido.ts
let client: KaleidoClient | null = null;

export function getClient(): KaleidoClient {
  if (!client) {
    client = new KaleidoClient({
      baseUrl: process.env.KALEIDO_API_URL!,
    });
  }
  return client;
}
```

```python Python
# lib/kaleido.py
_client = None

def get_client():
    global _client
    if _client is None:
        from kaleidoswap import KaleidoClient, KaleidoConfig
        config = KaleidoConfig(base_url=os.environ["KALEIDO_API_URL"])
        _client = KaleidoClient(config)
    return _client
```
</CodeGroup>

## Error Handling

### Always Catch Errors

<CodeGroup>
```typescript TypeScript
import { KaleidoClient, KaleidoError } from '@kaleidoswap/sdk';

try {
  const assets = await client.listAssets();
} catch (error) {
  if (error instanceof KaleidoError) {
    console.error('SDK error:', error.message);
  } else {
    throw error;  // Re-throw unexpected errors
  }
}
```

```python Python
from kaleidoswap import KaleidoError

try:
    assets = client.list_assets()
except KaleidoError as e:
    print(f"SDK error: {e}")
```
</CodeGroup>

### Use Specific Exception Types

Handle different errors appropriately:

<CodeGroup>
```typescript TypeScript
import { 
  ValidationError, 
  QuoteExpiredError, 
  NetworkError 
} from '@kaleidoswap/sdk';

try {
  const quote = await client.getQuoteByPair('BTC/USDT', amount);
} catch (error) {
  if (error instanceof ValidationError) {
    // User input error - show message
  } else if (error instanceof QuoteExpiredError) {
    // Get fresh quote
  } else if (error instanceof NetworkError) {
    // Retry with backoff
  }
}
```

```python Python
from kaleidoswap import ValidationError, QuoteExpiredError, NetworkError

try:
    quote = client.get_quote_by_pair("BTC/USDT", from_amount=amount)
except ValidationError as e:
    # User input error - show message
    pass
except QuoteExpiredError:
    # Get fresh quote
    pass
except NetworkError:
    # Retry with backoff
    pass
```
</CodeGroup>

## Async Patterns (TypeScript)

### Use Promise.all for Parallel Requests

```typescript
// ❌ Sequential (slow)
const assets = await client.listAssets();
const pairs = await client.listPairs();

// ✅ Parallel (fast)
const [assets, pairs] = await Promise.all([
  client.listAssets(),
  client.listPairs(),
]);
```

### Handle Partial Failures

```typescript
const results = await Promise.allSettled([
  client.getQuoteByPair('BTC/USDT', 1_000_000),
  client.getQuoteByPair('BTC/XAUT', 1_000_000),
]);

const successful = results
  .filter((r): r is PromiseFulfilledResult<Quote> => r.status === 'fulfilled')
  .map(r => r.value);
```

```typescript TypeScript

// Convert only for display

<CodeGroup>

    throw new Error(`Invalid amount: ${validation.errors.join(', ')}`);
  }

  const quote = await client.getQuoteByPair(pair, validation.rawAmount);
  // ... execute swap

```

## Caching

### Cache Static Data

Assets and pairs change infrequently:

<CodeGroup>
```typescript TypeScript
class MarketCache {
  private assets: Asset[] | null = null;
  private lastUpdate = 0;
  private ttl = 60000;  // 1 minute

  async getAssets(client: KaleidoClient): Promise<Asset[]> {
    if (!this.assets || Date.now() - this.lastUpdate > this.ttl) {
      this.assets = await client.listAssets();
      this.lastUpdate = Date.now();
    }
    return this.assets;
  }
}
```

```python Python
import time

class MarketCache:
    def __init__(self, ttl=60):
        self.assets = None
        self.last_update = 0
        self.ttl = ttl
    
    def get_assets(self, client):
        if self.assets is None or time.time() - self.last_update > self.ttl:
            self.assets = client.list_assets()
            self.last_update = time.time()
        return self.assets
```
</CodeGroup>

### Don't Cache Quotes

Quotes are time-sensitive - always fetch fresh:

```typescript
// ❌ Don't cache
const cachedQuote = cache.get('btc-usdt-quote');

// ✅ Always fetch fresh
const quote = await client.getQuoteByPair('BTC/USDT', amount);
```

## Security

### Never Log Sensitive Data

<CodeGroup>
```typescript TypeScript
// ❌ Don't log full responses
console.log('Quote:', JSON.stringify(quote));

// ✅ Log only necessary info
console.log(`Quote ${quote.rfqId}: ${quote.price}`);
```

```python Python
# ❌ Don't log full responses
print(f"Quote: {quote}")

# ✅ Log only necessary info
print(f"Quote {quote.rfq_id}: {quote.price}")
```
</CodeGroup>

### Validate External Input

```typescript
function validateTicker(ticker: string): boolean {
  return /^[A-Z0-9]{1,10}$/.test(ticker);
}

function validateAmount(amount: unknown): number | null {
  const num = Number(amount);
  if (isNaN(num) || num <= 0) return null;
  return num;
}
```
