## Overview

Both SDKs provide strong typing for all API responses:
- **TypeScript**: Native interfaces with full IDE autocomplete
- **Python**: Pydantic models with automatic validation

All types are auto-generated from OpenAPI specifications to ensure consistency.

## Model Serialization (Python)

Pydantic models can be easily serialized:

```python
# To dictionary
asset_dict = asset.model_dump()

# To JSON string
asset_json = asset.model_dump_json()

# From dictionary
asset = Asset.model_validate(asset_dict)

# From JSON string
import json
asset = Asset.model_validate(json.loads(asset_json))
```

## Type Guards (TypeScript)

Create type guards for runtime type checking:

```typescript
import { Asset, TradingPair } from '@kaleidoswap/sdk';

function isAsset(obj: unknown): obj is Asset {
  return typeof obj === 'object' && 
         obj !== null && 
         'ticker' in obj && 
         'precision' in obj;
}

function isTradingPair(obj: unknown): obj is TradingPair {
  return typeof obj === 'object' && 
         obj !== null && 
         'baseAsset' in obj && 
         'quoteAsset' in obj;
}
```
